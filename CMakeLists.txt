# Link to the official OBS runtime exports.
function(import_obs name dll)
  add_library(${name} SHARED IMPORTED GLOBAL)
  if(WIN32)
    get_filename_component(msvc_bin "${CMAKE_CXX_COMPILER}" DIRECTORY)
    find_program(dumpbin dumpbin HINTS "${msvc_bin}" REQUIRED)
    find_program(lib_tool lib HINTS "${msvc_bin}" REQUIRED)
    set(binary "${OBS_ROOT}/bin/64bit/${dll}.dll")
    execute_process(COMMAND "${dumpbin}" /nologo /exports "${binary}"
      OUTPUT_VARIABLE exports COMMAND_ERROR_IS_FATAL ANY)
    string(REPLACE "\n" ";" lines "${exports}")
    set(def "LIBRARY ${dll}.dll\nEXPORTS\n")
    set(count 0)
    foreach(line IN LISTS lines)
      if(line MATCHES "^[ \t]+[0-9]+[ \t]+[0-9A-Fa-f]+[ \t]+[0-9A-Fa-f]+[ \t]+([A-Za-z_][A-Za-z_0-9]*)")
        string(APPEND def "${CMAKE_MATCH_1}\n")
        math(EXPR count "${count}+1")
      endif()
    endforeach()
    if(count LESS 20)
      message(FATAL_ERROR "Cannot read exports from ${binary}")
    endif()
    set(base "${CMAKE_CURRENT_BINARY_DIR}/obs-sdk/${dll}")
    file(WRITE "${base}.def" "${def}")
    execute_process(COMMAND "${lib_tool}" /nologo /machine:x64 "/def:${base}.def" "/out:${base}.lib"
      OUTPUT_QUIET COMMAND_ERROR_IS_FATAL ANY)
    set_target_properties(${name} PROPERTIES IMPORTED_LOCATION "${binary}" IMPORTED_IMPLIB "${base}.lib")
  else()
    find_library(found_${dll} NAMES ${dll}.1.dylib ${dll}.dylib ${dll} lib${dll}
      PATHS "${OBS_ROOT}/Contents/Frameworks" NO_DEFAULT_PATH REQUIRED)
    set(binary "${found_${dll}}")
    if(IS_DIRECTORY "${binary}")
      get_filename_component(stem "${binary}" NAME_WE)
      set(binary "${binary}/${stem}")
    endif()
    set_target_properties(${name} PROPERTIES IMPORTED_LOCATION "${binary}")
  endif()
endfunction()

  target_sources(obs-docksnap PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/version.rc")
  target_compile_options(obs-docksnap PRIVATE /W4 /WX /utf-8 /permissive-)
  target_link_options(obs-docksnap PRIVATE /INCREMENTAL:NO /DEBUG:NONE)
  install(TARGETS obs-docksnap RUNTIME DESTINATION obs-docksnap/bin/64bit LIBRARY DESTINATION obs-docksnap/bin/64bit)
  install(DIRECTORY data/ DESTINATION obs-docksnap/data)
  install(FILES LICENSE DESTINATION obs-docksnap)
  install(FILES scripts/install_OBS-DockSnap.bat DESTINATION .)
